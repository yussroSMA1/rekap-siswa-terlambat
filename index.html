Hubungkan Langsung ke Google Sheets

Anda dapat mengirimkan data rekap secara otomatis dari aplikasi web ini langsung ke spreadsheet Google Sheets sekolah Anda menggunakan Google Apps Script Web App.

Realtime Auto Sync

1 Masukkan Webhook URL Google Apps Script

Tempelkan URL Web App dari Apps Script yang telah di-deploy dari Google Sheet Anda:

Google Apps Script (Code.gs)

Salin kode berikut ke Google Apps Script (Extensions -> Apps Script pada Google Sheets Anda) lalu klik Deploy -> New Deployment -> Anyone can access. 

function doPost(e) {
  try {
    var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet();
    var data = JSON.parse(e.postData.contents);
    
    // Header jika sheet masih kosong
    if (sheet.getLastRow() === 0) {
      sheet.appendRow(["ID", "Tanggal", "Waktu", "Nama Siswa", "Kelas", "Alasan", "Sanksi/Tindak Lanjut", "Petugas"]);
    }
    
    // Tambah Baris Rekap Siswa Terlambat
    sheet.appendRow([
      data.id || "",
      data.tanggal || "",
      data.waktu || "",
      data.nama || "",
      data.kelas || "",
      data.alasan || "",
      data.sanksi || "",
      data.petugas || ""
    ]);
    
    return ContentService.createTextOutput(JSON.stringify({"status": "success"}))
      .setMimeType(ContentService.MimeType.JSON);
  } catch (err) {
    return ContentService.createTextOutput(JSON.stringify({"status": "error", "message": err.toString()}))
      .setMimeType(ContentService.MimeType.JSON);
  }
}
